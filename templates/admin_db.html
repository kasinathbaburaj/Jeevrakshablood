SQLite Relational DDL Architecture

The active database schema with primary keys, foreign keys, cascade constraints, and indices:

-- 1. Users Identity & Credentials CREATE TABLE users ( id INTEGER PRIMARY KEY AUTOINCREMENT, full_name TEXT NOT NULL, email TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, phone TEXT UNIQUE NOT NULL, blood_group TEXT NOT NULL, city TEXT NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- 2. Registered Active Donors CREATE TABLE donors ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER UNIQUE NOT NULL, blood_group TEXT NOT NULL, age INTEGER NOT NULL, weight REAL NOT NULL, last_donation_date DATE, is_available INTEGER DEFAULT 1, medical_verified INTEGER DEFAULT 1, location TEXT NOT NULL, FOREIGN KEY(user_id) REFERENCES users(id) ON DELETE CASCADE ); -- 3. Emergency Blood Requisitions CREATE TABLE blood_requests ( id INTEGER PRIMARY KEY AUTOINCREMENT, requester_id INTEGER NOT NULL, patient_name TEXT NOT NULL, blood_group_needed TEXT NOT NULL, units_required INTEGER NOT NULL, urgency_level TEXT NOT NULL, hospital_name TEXT NOT NULL, location TEXT NOT NULL, contact_phone TEXT NOT NULL, status TEXT DEFAULT 'Pending', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY(requester_id) REFERENCES users(id) ON DELETE CASCADE ); -- 4. Donation Log & Match Audit CREATE TABLE donations_log ( id INTEGER PRIMARY KEY AUTOINCREMENT, donor_id INTEGER NOT NULL, request_id INTEGER, donation_date DATE DEFAULT CURRENT_DATE, units_donated INTEGER DEFAULT 1, status TEXT DEFAULT 'Completed', FOREIGN KEY(donor_id) REFERENCES donors(id) ON DELETE CASCADE, FOREIGN KEY(request_id) REFERENCES blood_requests(id) ON DELETE SET NULL );
